#!/usr/bin/env python3
"""Insere os blocos do mcp-server nos compose files da VPS (idempotente)."""
import shutil
import sys

MCP_SERVICE_BLOCK = """  # ── MCP Server (Node/Express) ─────────────────────────────────────────────
  # Cliente do api-gateway apenas — nunca acessa Prisma/DB diretamente (D2 do
  # SDD de integração MCP). Autenticação é por PAT (Authorization: Bearer),
  # repassado ao gateway a cada chamada; nenhum INTERNAL_API_KEY é necessário aqui.
  mcp-server:
    image: ghcr.io/${GHCR_OWNER}/mcp-server:${IMAGE_TAG}
    restart: always
    security_opt:
      - no-new-privileges:true
    environment:
      - NODE_ENV=${NODE_ENV}
      - API_GATEWAY_INTERNAL_URL=http://api-gateway:4000
      - PORT=4006
    labels:
      - "traefik.enable=true"
      - "traefik.http.services.mcp-server-${COMPOSE_PROJECT_NAME}.loadbalancer.server.port=4006"
    healthcheck:
      test: ["CMD-SHELL", "wget -qO- http://localhost:4006/health || exit 1"]
      interval: 15s
      timeout: 5s
      retries: 3
      start_period: 15s
    networks:
      - proxy
      - internal
    depends_on:
      api-gateway:
        condition: service_healthy

"""

MIGRATE_AUTH_BLOCK = """  # Mantido por paridade com os demais serviços e para uso manual/CI pontual.
  # A automação real do deploy vem do docker-entrypoint.sh do auth-service,
  # que roda `prisma migrate deploy` no boot do container (o webhook de
  # deploy só executa `up -d`, sem o profile `migration`).
  migrate-auth-service:
    image: ghcr.io/${GHCR_OWNER}/auth-service:${IMAGE_TAG}
    command: prisma migrate deploy
    read_only: true
    tmpfs:
      - /tmp
    security_opt:
      - no-new-privileges:true
    env_file:
      - .env
    networks:
      - internal
    depends_on:
      postgres:
        condition: service_healthy
    restart: "no"
    profiles:
      - migration

"""

MCP_PROD_OVERRIDE_BLOCK = """  mcp-server:
    image: ghcr.io/${GHCR_OWNER}/mcp-server:prod
    deploy:
      resources:
        limits:
          memory: 256M
    env_file:
      - .env
    networks:
      - traefik-public
    labels:
      - "traefik.enable=true"
      - "traefik.http.routers.mcp-prod.rule=Host(`mcp.operum.adm.br`)"
      - "traefik.http.routers.mcp-prod.entrypoints=websecure"
      - "traefik.http.routers.mcp-prod.tls.certresolver=myresolver"
      - "traefik.http.routers.mcp-prod.service=mcp-server-${COMPOSE_PROJECT_NAME}"
      - "traefik.http.routers.mcp-prod.middlewares=secure-headers@file"

"""


def insert_before(path: str, anchor: str, block: str, marker: str) -> bool:
    with open(path, encoding="utf-8") as f:
        content = f.read()

    if marker in content:
        print(f"[skip] {path}: já contém '{marker.strip()}', nada a fazer")
        return False

    if anchor not in content:
        print(f"[ERRO] {path}: âncora não encontrada:\n{anchor!r}")
        sys.exit(1)

    shutil.copy(path, path + ".bak")
    content = content.replace(anchor, block + anchor, 1)
    with open(path, "w", encoding="utf-8") as f:
        f.write(content)
    print(f"[ok] {path}: bloco inserido (backup em {path}.bak)")
    return True


insert_before(
    "docker-compose.yml",
    "  # ── Auth Service (NestJS) ──────────────────────────────────────────────────\n",
    MCP_SERVICE_BLOCK,
    "\n  mcp-server:\n",
)

insert_before(
    "docker-compose.yml",
    "  # ── PostgreSQL ─────────────────────────────────────────────────────────────\n",
    MIGRATE_AUTH_BLOCK,
    "\n  migrate-auth-service:\n",
)

insert_before(
    "docker-compose.production.yml",
    "  auth-service:\n",
    MCP_PROD_OVERRIDE_BLOCK,
    "\n  mcp-server:\n",
)

print("Concluído.")
